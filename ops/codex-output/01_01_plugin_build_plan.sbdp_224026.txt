namespace BSP\Core;

final class Modules
{
    /** @var ModuleInterface[] */
    private array $modules = [];

    public function register(ModuleInterface $module): void
    {
        $this->modules[] = $module;
    }

    public function boot(): void
    {
        foreach ($this->modules as $module) {
            $module->init();
        }
    }
}

<?php

declare(strict_types=1);

namespace BSP\Notifications;

use WP_REST_Request;
use WP_REST_Response;

final class RestController
{
    public function register_routes(): void
    {
        register_rest_route(
            'bsp/v1',
            '/notifications',
            [
                'methods'             => \WP_REST_Server::READABLE,
                'callback'            => [$this, 'get_items'],
                'permission_callback' => [$this, 'permissions_check'],
            ]
        );
    }

    public function permissions_check(WP_REST_Request $request): bool
    {
        return current_user_can('read');
    }

    public function get_items(WP_REST_Request $request): WP_REST_Response
    {
        return rest_ensure_response(['ok' => true]);
    }
}
```

            '<div class="%s">%s</div>',
            esc_attr($wrapper_class),
            $list_html

<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

final class Module implements ModuleInterface
{
    private RestController $restController;

    public function __construct(?RestController $restController = null)
    {
        $this->restController = $restController ?? new RestController();
    }

    public function init(): void
    {
        add_action('rest_api_init', [$this, 'register_rest_routes']);
        add_shortcode('booking_notifications', [$this, 'render_shortcode']);
        add_action('admin_menu', [$this, 'register_admin_page']);
        add_action('admin_init', [$this, 'register_admin_settings']);
    }

    public function register_rest_routes(): void
    {
        $this->restController->register_routes();
    }

    public function render_shortcode(): string
    {
        $wrapper_class = 'bsp-notifications';
        $list_html = sprintf(
            '<ul class="%s" data-role="%s"></ul>',
            esc_attr('bsp-notifications__list'),
            esc_attr('notification-list')
        );

        $content = sprintf(
            '<div class="%s">%s</div>',
            esc_attr($wrapper_class),
            $list_html
        );

        return wp_kses_post($content);
    }

    public function register_admin_page(): void
    {
        add_options_page(
            __('Booking Pro', 'sbdp'),
            __('Booking Pro', 'sbdp'),
            'manage_options',
            'booking-pro-settings',
            [$this, 'render_settings_page']
        );
    }

    public function render_settings_page(): void
    {
        if (!current_user_can('manage_options')) {
            wp_die(esc_html__('You do not have permission to access this page.', 'sbdp'));
        }

        echo '<div class="wrap">';
        echo '<h1>' . esc_html__('Booking Pro', 'sbdp') . '</h1>';
        echo '<form action="options.php" method="post">';
        settings_fields('bsp_notifications');
        do_settings_sections('booking-pro-settings');
        submit_button();
        echo '</form>';
        echo '</div>';
    }

    public function register_admin_settings(): void
    {
        register_setting(
            'bsp_notifications',
            'bsp_notifications_enabled',
            [
                'type' => 'boolean',
                'sanitize_callback' => [$this, 'sanitize_enabled_option'],
                'default' => true,
            ]
        );

        add_settings_section(
            'bsp_notifications_section',
            __('Notifications', 'sbdp'),
            [$this, 'render_settings_section'],
            'booking-pro-settings'
        );

        add_settings_field(
            'bsp_notifications_enabled',
            __('Enable Notifications', 'sbdp'),
            [$this, 'render_enabled_field'],
            'booking-pro-settings',
            'bsp_notifications_section'
        );
    }

    public function render_settings_section(): void
    {
        echo '<p>' . esc_html__('Configure notification delivery for Booking Pro.', 'sbdp') . '</p>';
    }

    public function render_enabled_field(): void
    {
        $value = (bool) get_option('bsp_notifications_enabled', true);

        echo '<label for="bsp_notifications_enabled">';
        echo '<input type="checkbox" name="bsp_notifications_enabled" id="bsp_notifications_enabled" value="1" ' . checked($value, true, false) . ' />';
        echo '&nbsp;' . esc_html__('Enable notifications module', 'sbdp');
        echo '</label>';
    }

    /**
     * @param mixed $value Raw option input.
     */
    public function sanitize_enabled_option($value): bool
    {
        return rest_sanitize_boolean($value);
    }
}
```
