```php
<?php

declare(strict_types=1);

namespace BSP\Core;

final class Modules
{
    /**
     * @var ModuleInterface[]
     */
    private array $items = [];

    public function register(ModuleInterface $module): void
    {
        $this->items[] = $module;
        $module->register();
    }

    public function boot(): void
    {
        foreach ($this->items as $module) {
            $module->boot();
            $module->init();
        }
    }

    /**
     * @return ModuleInterface[]
     */
    public function all(): array
    {
        return $this->items;
    }
}
```
- `core/src/CoreServiceProvider.php`:
```php
<?php

declare(strict_types=1);

namespace BSP\Core;

final class CoreServiceProvider
{
    private Modules $modules;

    public function __construct(?Modules $modules = null)
    {
        $this->modules = $modules ?? new Modules();
    }

    public function modules(): Modules
    {
        return $this->modules;
    }

    public function boot(): void
    {
        do_action('bsp/core/service_provider/booting', $this->modules);
    }
}
```

```php
<?php

declare(strict_types=1);
```

```php
namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

final class Module implements ModuleInterface
{
    public function register(): void
    {
    }

    public function boot(): void
    {
    }

    public function init(): void
    {
        add_action('rest_api_init', [$this, 'register_rest_routes']);
        add_shortcode('booking_notifications', [$this, 'render_shortcode']);
        add_action('admin_menu', [$this, 'register_admin_page']);
        add_action('admin_init', [$this, 'register_settings']);
    }

    public function register_rest_routes(): void
    {
        $controller = new RestController();
        $controller->register_routes();
    }

    public function render_shortcode(): string
    {
        $content = '<div class="sbdp-notifications">';
        $content .= '<h2>' . esc_html__('Booking Notifications', 'sbdp') . '</h2>';
        $content .= '<ul class="sbdp-notifications__list">';
        $content .= '<li class="sbdp-notifications__item">' . esc_html__('Notifications will appear here.', 'sbdp') . '</li>';
        $content .= '</ul></div>';

        return wp_kses_post($content);
    }

    public function register_admin_page(): void
    {
        add_options_page(
            __('Booking Pro Settings', 'sbdp'),
            __('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-booking-pro',
            [$this, 'render_settings_page']
        );
    }

    public function render_settings_page(): void
    {
        if (! current_user_can('manage_options')) {
            return;
        }
        ?>
        <div class="wrap">
            <h1><?php echo esc_html__('Booking Pro Settings', 'sbdp'); ?></h1>
            <form action="options.php" method="post">
                <?php
                settings_fields('bsp_notifications');
                do_settings_sections('bsp-booking-pro');
                submit_button(__('Save Changes', 'sbdp'));
                ?>
            </form>
        </div>
        <?php
    }

    public function register_settings(): void
    {
        register_setting(
            'bsp_notifications',
            'bsp_notifications_enabled',
            [
                'type' => 'boolean',
                'default' => false,
                'sanitize_callback' => static fn($value): bool => (bool) $value,
            ]
        );

        add_settings_section(
            'bsp_notifications_section',
            __('Notifications', 'sbdp'),
            static function (): void {
                echo '<p>' . esc_html__('Manage notification settings for Booking Pro Suite.', 'sbdp') . '</p>';
            },
            'bsp-booking-pro'
        );

        add_settings_field(
            'bsp_notifications_enabled',
            __('Enable notifications', 'sbdp'),
            [$this, 'render_notifications_enabled_field'],
            'bsp-booking-pro',
            'bsp_notifications_section'
        );
    }

    public function render_notifications_enabled_field(): void
    {
        $enabled = (bool) get_option('bsp_notifications_enabled', false);
        ?>
        <label>
            <input type="checkbox" name="bsp_notifications_enabled" value="1" <?php checked($enabled); ?> />
            <?php echo esc_html__('Enable notifications for new bookings.', 'sbdp'); ?>
        </label>
        <?php
    }
}
```

```php
<?php

declare(strict_types=1);

namespace BSP\Notifications;

use WP_REST_Controller;
use WP_REST_Request;
use WP_REST_Server;

final class RestController extends WP_REST_Controller
{
    protected string $namespace = 'bsp/v1';
    protected string $rest_base = 'notifications';

    public function register_routes(): void
    {
        register_rest_route(
            $this->namespace,
            '/' . $this->rest_base,
            [
                [
                    'methods'             => WP_REST_Server::READABLE,
                    'callback'            => [$this, 'get_items'],
                    'permission_callback' => [$this, 'get_items_permissions_check'],
                ],
            ]
        );
    }

    public function get_items(WP_REST_Request $request)
    {
        return rest_ensure_response(
            [
                'ok' => true,
            ]
        );
    }

    public function get_items_permissions_check(WP_REST_Request $request): bool
    {
        return current_user_can('read');
    }
}
```

```php
namespace BSP\Tests\Notifications;

use Brain\Monkey\Actions;

    protected function setUp(): void
    {
        parent::setUp();
        \Brain\Monkey\setUp();
    }

    protected function tearDown(): void
    {
        \Brain\Monkey\tearDown();
        parent::tearDown();
    }

    public function test_init_registers_expected_hooks(): void
    {
        $module = new Module();

        Actions::expectAdded('rest_api_init')->once()->with([$module, 'register_rest_routes']);
        Functions::expect('add_shortcode')->once()->with('booking_notifications', [$module, 'render_shortcode']);
        Actions::expectAdded('admin_menu')->once()->with([$module, 'register_admin_page']);
        Actions::expectAdded('admin_init')->once()->with([$module, 'register_settings']);

        $module->init();
    }
}
```

```bash
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"

pushd "$ROOT" >/dev/null

composer install --no-dev

find core modules -type f -name '*.php' -print0 | xargs -0 -n1 php -l

mkdir -p dist
zip -r "dist/booking-pro-suite.zip" \

    core \
    modules \
    assets \
    languages \
```
