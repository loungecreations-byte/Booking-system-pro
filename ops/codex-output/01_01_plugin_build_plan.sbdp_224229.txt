# File: modules/notifications/Module.php
```php
<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

use function add_action;
use function add_options_page;
use function add_settings_field;
use function add_settings_section;
use function add_shortcode;
use function admin_url;
use function checked;
use function current_user_can;
use function do_settings_sections;
use function esc_attr;
use function esc_html__;
use function esc_url;
use function get_option;
use function register_setting;
use function settings_fields;
use function submit_button;
use function wp_die;
use function wp_kses_post;

final class Module implements ModuleInterface
{
    private const SHORTCODE_TAG = 'booking_notifications';
    private const OPTION_KEY = 'bsp_notifications_enabled';

    private NotificationsRestController $restController;

    public function __construct(?NotificationsRestController $restController = null)
    {
        $this->restController = $restController ?? new NotificationsRestController();
    }

    public function init(): void
    {
        add_action('rest_api_init', array($this->restController, 'register_routes'));
        add_action('admin_menu', array($this, 'register_admin_page'));
        add_action('admin_init', array($this, 'register_settings'));
        add_shortcode(self::SHORTCODE_TAG, array($this, 'render_shortcode'));
    }

    public function boot(): void
    {
        // Additional boot logic can be added here.
    }

    public function register_admin_page(): void
    {
        add_options_page(
            esc_html__('Booking Pro Notifications', 'sbdp'),
            esc_html__('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-notifications',
            array($this, 'render_settings_page')
        );
    }

    public function register_settings(): void
    {
        register_setting(
            'bsp_notifications',
            self::OPTION_KEY,
            array(
                'type'              => 'boolean',
                'sanitize_callback' => array($this, 'sanitize_checkbox'),
                'default'           => true,
            )
        );

        add_settings_section(
            'bsp_notifications_section',
            esc_html__('Notifications', 'sbdp'),
            array($this, 'render_section_intro'),
            'bsp-notifications'
        );

        add_settings_field(
            self::OPTION_KEY,
            esc_html__('Enable shortcode output', 'sbdp'),
            array($this, 'render_enable_field'),
            'bsp-notifications',
            'bsp_notifications_section'
        );
    }

    public function render_settings_page(): void
    {
        if (! current_user_can('manage_options')) {
            wp_die(esc_html__('You do not have permission to manage these settings.', 'sbdp'));
        }

        echo '<div class="wrap">';
        echo '<h1>' . esc_html__('Booking Pro Notifications', 'sbdp') . '</h1>';
        echo '<form action="' . esc_url(admin_url('options.php')) . '" method="post">';
        settings_fields('bsp_notifications');
        do_settings_sections('bsp-notifications');
        submit_button();
        echo '</form>';
        echo '</div>';
    }

    public function render_section_intro(): void
    {
        echo '<p>' . esc_html__('Toggle the booking notifications shortcode output.', 'sbdp') . '</p>';
    }

    public function render_enable_field(): void
    {
        $enabled = (bool) get_option(self::OPTION_KEY, true);

        echo '<label for="' . esc_attr(self::OPTION_KEY) . '">';
        echo '<input type="checkbox" id="' . esc_attr(self::OPTION_KEY) . '" name="' . esc_attr(self::OPTION_KEY) . '" value="1" ' . checked($enabled, true, false) . ' />';
        echo ' ' . esc_html__('Enable the [booking_notifications] shortcode output.', 'sbdp');
        echo '</label>';
    }

    /**
     * @param mixed $value Raw option value.
     */
    public function sanitize_checkbox($value): bool
    {
        if (is_bool($value)) {
            return $value;
        }

        if (is_string($value)) {
            $normalized = strtolower(trim($value));
            return in_array($normalized, array('1', 'true', 'on', 'yes'), true);
        }

        if (is_int($value)) {
            return 1 === $value;
        }

        return false;
    }

    /**
     * @param array<string, mixed> $atts
     * @param string|null $content
     */
    public function render_shortcode(array $atts = array(), ?string $content = null): string
    {
        $enabled = (bool) get_option(self::OPTION_KEY, true);

        if (! $enabled) {
            return '';
        }

        $placeholder = esc_html__('Notifications will appear here.', 'sbdp');

        $markup = sprintf(
            '<div class="bsp-notifications" data-component="%s" aria-live="polite"><ul class="bsp-notifications__list"><li class="bsp-notifications__item">%s</li></ul></div>',
            esc_attr('booking-notifications'),
            $placeholder
        );

        return (string) wp_kses_post($markup);
    }
}
