<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

final class Module implements ModuleInterface
{
    private NotificationsRestController $restController;

    public function __construct(?NotificationsRestController $restController = null)
    {
        $this->restController = $restController ?? new NotificationsRestController();
    }

    public function init(): void
    {
        \add_action('rest_api_init', [$this, 'register_rest_routes']);
        \add_shortcode('booking_notifications', [$this, 'render_shortcode']);
        \add_action('admin_menu', [$this, 'register_settings_page']);
        \add_action('admin_init', [$this, 'register_settings']);
    }

    public function boot(): void
    {
        // Boot logic reserved for future use.
    }

    public function register_rest_routes(): void
    {
        $this->restController->register_routes();
    }

    public function render_shortcode($atts = [], $content = null): string
    {
        $heading = \esc_html__('Booking Notifications', 'sbdp');
        $placeholder = \esc_html__('Notifications will appear here once available.', 'sbdp');

        $markup = sprintf(
            '<div class="bsp-notifications" aria-live="polite"><h2 class="bsp-notifications__heading">%1$s</h2><ul class="bsp-notifications__list"><li class="bsp-notifications__item">%2$s</li></ul></div>',
            $heading,
            $placeholder
        );

        return \wp_kses_post($markup);
    }

    public function register_settings_page(): void
    {
        \add_options_page(
            \esc_html__('Booking Pro Settings', 'sbdp'),
            \esc_html__('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-booking-pro',
            [$this, 'render_settings_page']
        );
    }

    public function register_settings(): void
    {
        \register_setting(
            'bsp_notifications',
            'bsp_notifications_enabled',
            [
                'type' => 'string',
                'sanitize_callback' => [$this, 'sanitize_enabled_option'],
                'default' => 'no',
            ]
        );

        \add_settings_section(
            'bsp_notifications_main',
            \esc_html__('Notifications Settings', 'sbdp'),
            static function (): void {
                echo '<p>' . \esc_html__('Configure Booking Pro notification defaults.', 'sbdp') . '</p>';
            },
            'bsp_notifications'
        );

        \add_settings_field(
            'bsp_notifications_enabled',
            \esc_html__('Enable Notifications', 'sbdp'),
            [$this, 'render_notifications_enabled_field'],
            'bsp_notifications',
            'bsp_notifications_main'
        );
    }

    public function render_settings_page(): void
    {
        if (! \current_user_can('manage_options')) {
            return;
        }

        echo '<div class="wrap">';
        echo '<h1>' . \esc_html__('Booking Pro Suite', 'sbdp') . '</h1>';
        echo '<form method="post" action="options.php">';
        \settings_fields('bsp_notifications');
        \do_settings_sections('bsp_notifications');
        \submit_button();
        echo '</form>';
        echo '</div>';
    }

    public function render_notifications_enabled_field(): void
    {
        $current = \get_option('bsp_notifications_enabled', 'no');
        $checked = \checked($current, 'yes', false);

        echo '<label for="bsp_notifications_enabled">';
        echo '<input type="checkbox" id="bsp_notifications_enabled" name="bsp_notifications_enabled" value="yes" ' . $checked . ' />';
        echo ' ' . \esc_html__('Enable email and in-app notifications for bookings.', 'sbdp');
        echo '</label>';
    }

    public function sanitize_enabled_option($value): string
    {
        return $value === 'yes' ? 'yes' : 'no';
    }
}
