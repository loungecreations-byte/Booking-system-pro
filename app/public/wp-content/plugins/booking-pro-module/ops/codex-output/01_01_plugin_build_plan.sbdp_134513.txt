```php
<?php

declare(strict_types=1);

namespace BSP\Core;

use function add_action;
use function function_exists;
use function get_class;

class Modules
{
    /**
     * @var array<string, ModuleInterface>
     */
    private array $modules = [];

    public function register(ModuleInterface $module): void
    {
        $this->modules[get_class($module)] = $module;
        $module->register();
    }

    public function boot(): void
    {
        foreach ($this->modules as $module) {
            $module->boot();

            if (function_exists('add_action')) {
                add_action('init', [$module, 'init']);
            } else {
                $module->init();
            }
        }
    }

    /**
     * @return array<string, ModuleInterface>
     */
    public function all(): array
    {
        return $this->modules;
    }
}
```

core/src/CoreServiceProvider.php
```php
<?php

declare(strict_types=1);

namespace BSP\Core;

use function dirname;
use function function_exists;
use function load_plugin_textdomain;
use function plugin_basename;

class CoreServiceProvider
{
    private string $pluginFile;

    public function __construct(string $pluginFile)
    {
        $this->pluginFile = $pluginFile;
    }

    public function boot(): void
    {
        $this->loadTextdomain();
    }

    private function loadTextdomain(): void
    {
        if (! function_exists('load_plugin_textdomain') || ! function_exists('plugin_basename')) {
            return;
        }

        load_plugin_textdomain(
            'sbdp',
            false,
            dirname(plugin_basename($this->pluginFile)) . '/languages'
        );
    }
}
```

```php
<?php

declare(strict_types=1);

namespace BSP\Notifications;

use WP_REST_Controller;
use WP_REST_Request;
use WP_REST_Response;
use WP_REST_Server;

class NotificationsRestController extends WP_REST_Controller
{
    public function __construct()
    {
        $this->namespace = 'bsp/v1';
        $this->rest_base = 'notifications';
    }

    public function register_routes(): void
    {
        register_rest_route(
            $this->namespace,
            '/' . $this->rest_base,
            [
                [
                    'methods' => WP_REST_Server::READABLE,
                    'callback' => [$this, 'get_items'],
                    'permission_callback' => [$this, 'get_items_permissions_check'],
                ],
            ]
        );
    }

    public function get_items(WP_REST_Request $request): WP_REST_Response
    {
        return new WP_REST_Response(['ok' => true], 200);
    }

    public function get_items_permissions_check(WP_REST_Request $request): bool
    {
        if (function_exists('current_user_can')) {
            return current_user_can('read');
        }

        return true;
    }
```
```php
}

    public function register(): void
    {
        add_action('rest_api_init', [$this->restController, 'register_routes']);
    }

    public function boot(): void
    {
        // Additional boot logic will be introduced in later steps.
    }

    public function init(): void
    {
        add_shortcode('booking_notifications', [$this, 'renderNotificationsShortcode']);
    }

    public function renderNotificationsShortcode(): string
    {
        $enabled = get_option('bsp_notifications_enabled', 'yes');

        if ('yes' !== $enabled) {
            $message = sprintf(
                '<div class="bsp-notifications bsp-notifications--disabled">%s</div>',
                esc_html__('Booking notifications are currently disabled.', 'sbdp')
            );

            return wp_kses_post($message);
        }

        ob_start();
        ?>
        <div class="bsp-notifications" data-enabled="<?php echo esc_attr($enabled); ?>">
            <h2 class="bsp-notifications__title"><?php echo esc_html__('Booking Notifications', 'sbdp'); ?></h2>
            <ul class="bsp-notifications__list" aria-live="polite">
                <li class="bsp-notifications__item"><?php echo esc_html__('No notifications yet.', 'sbdp'); ?></li>
            </ul>
        </div>
        <?php
        $html = ob_get_clean();

        return is_string($html) ? wp_kses_post($html) : '';
    }
```

```php
<?php

declare(strict_types=1);

namespace BSP\Notifications;
```

```php
use BSP\Core\ModuleInterface;

class Module implements ModuleInterface
{
    private const OPTION_ENABLED = 'bsp_notifications_enabled';

    private NotificationsRestController $restController;

    public function __construct()
    {
        $this->restController = new NotificationsRestController();
    }

    public function register(): void
    {
        add_action('rest_api_init', [$this->restController, 'register_routes']);
        add_action('admin_menu', [$this, 'registerAdminPage']);
        add_action('admin_init', [$this, 'registerSettings']);
    }

    public function boot(): void
    {
        if (null === get_option(self::OPTION_ENABLED, null)) {
            update_option(self::OPTION_ENABLED, 'yes');
        }
    }

    public function init(): void
    {
        add_shortcode('booking_notifications', [$this, 'renderNotificationsShortcode']);
    }

    public function renderNotificationsShortcode(): string
    {
        $enabled = get_option(self::OPTION_ENABLED, 'yes');

        if ('yes' !== $enabled) {
            $message = sprintf(
                '<div class="bsp-notifications bsp-notifications--disabled">%s</div>',
                esc_html__('Booking notifications are currently disabled.', 'sbdp')
            );

            return wp_kses_post($message);
        }

        ob_start();
        ?>
        <div class="bsp-notifications" data-enabled="<?php echo esc_attr($enabled); ?>">
            <h2 class="bsp-notifications__title"><?php echo esc_html__('Booking Notifications', 'sbdp'); ?></h2>
            <ul class="bsp-notifications__list" aria-live="polite">
                <li class="bsp-notifications__item"><?php echo esc_html__('No notifications yet.', 'sbdp'); ?></li>
            </ul>
        </div>
        <?php
        $html = ob_get_clean();

        return is_string($html) ? wp_kses_post($html) : '';
    }

    public function registerAdminPage(): void
    {
        add_options_page(
            __('Booking Pro Notifications', 'sbdp'),
            __('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-notifications',
            [$this, 'renderSettingsPage']
        );
    }

    public function registerSettings(): void
    {
        register_setting(
            'bsp_notifications',
            self::OPTION_ENABLED,
            [
                'type' => 'string',
                'sanitize_callback' => [$this, 'sanitizeEnabledOption'],
                'default' => 'yes',
            ]
        );

        add_settings_section(
            'bsp_notifications_general',
            __('Notifications Settings', 'sbdp'),
            [$this, 'renderSettingsSection'],
            'bsp_notifications'
        );

        add_settings_field(
            self::OPTION_ENABLED,
            __('Enable Notifications', 'sbdp'),
            [$this, 'renderNotificationsEnabledField'],
            'bsp_notifications',
            'bsp_notifications_general'
        );
    }

    public function renderSettingsSection(): void
    {
        echo '<p>' . esc_html__('Control the Booking Pro Suite notification features.', 'sbdp') . '</p>';
    }

    public function renderNotificationsEnabledField(): void
    {
        $value = get_option(self::OPTION_ENABLED, 'yes');
        ?>
        <label for="<?php echo esc_attr(self::OPTION_ENABLED); ?>">
            <input
                type="checkbox"
                id="<?php echo esc_attr(self::OPTION_ENABLED); ?>"
                name="<?php echo esc_attr(self::OPTION_ENABLED); ?>"
                value="yes"
                <?php checked('yes', $value); ?>
            />
            <?php echo esc_html__('Notify planners about important booking events.', 'sbdp'); ?>
        </label>
        <?php
    }

    public function renderSettingsPage(): void
    {
        if (! current_user_can('manage_options')) {
            wp_die(esc_html__('You do not have permission to access this page.', 'sbdp'));
        }
        ?>
        <div class="wrap">
            <h1><?php echo esc_html__('Booking Pro Suite Notifications', 'sbdp'); ?></h1>
            <form method="post" action="options.php">
                <?php
                settings_fields('bsp_notifications');
                do_settings_sections('bsp_notifications');
                submit_button();
                ?>
            </form>
        </div>
        <?php
    }

    /**
     * @param mixed $value Raw option value.
     */
    public function sanitizeEnabledOption($value): string
    {
        return 'yes' === $value ? 'yes' : 'no';
    }
}
```

```php
            return current_user_can('read');

        return true;
    }
```

```php
<?php

declare(strict_types=1);

namespace BSP\Core;

use function add_action;
use function function_exists;
use function get_class;

class Modules
{
    /**
     * @var array<string, ModuleInterface>
     */
    private array $modules = [];

    public function register(ModuleInterface $module): void
    {
        $this->modules[get_class($module)] = $module;
        $module->register();
    }

    public function boot(): void
    {
        foreach ($this->modules as $module) {
            $module->boot();

            if (function_exists('add_action')) {
                add_action('init', [$module, 'init']);
            } else {
                $module->init();
            }
        }
    }

    /**
     * @return array<string, ModuleInterface>
     */
    public function all(): array
    {
        return $this->modules;
    }
}
```

```bash
#!/usr/bin/env bash
set -euo pipefail

ROOT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
DIST_DIR="${ROOT_DIR}/dist"
BUILD_ROOT="$(mktemp -d)"
STAGING_DIR="${BUILD_ROOT}/booking-pro-suite"

cleanup() {
    rm -rf "${BUILD_ROOT}"
}
trap cleanup EXIT

mkdir -p "${STAGING_DIR}"

if command -v rsync >/dev/null 2>&1; then
    rsync -a \
        --exclude '.git/' \
        --exclude 'dist/' \
        --exclude 'node_modules/' \
        --exclude 'vendor/' \
        --exclude '*.log' \
        "${ROOT_DIR}/" \
        "${STAGING_DIR}/"
else
    (cd "${ROOT_DIR}" && tar --exclude='./dist' --exclude='./vendor' --exclude='./.git' -cf - .) | (cd "${STAGING_DIR}" && tar -xf -)
fi

composer install --no-dev --prefer-dist --working-dir="${STAGING_DIR}"

find "${ROOT_DIR}" -type f -name '*.php' -not -path "${ROOT_DIR}/vendor/*" -not -path "${ROOT_DIR}/dist/*" -print0 | xargs -0 -n1 php -l

mkdir -p "${DIST_DIR}"
(
    cd "${BUILD_ROOT}"
    zip -rq "${DIST_DIR}/booking-pro-suite.zip" "booking-pro-suite"
)

echo "Package created at ${DIST_DIR}/booking-pro-suite.zip"
```
