<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

final class Module implements ModuleInterface
{
    private const SHORTCODE_TAG = 'booking_notifications';

    private Rest_Controller $restController;

    private Admin_Settings_Page $adminSettingsPage;

    public function __construct(?Rest_Controller $restController = null, ?Admin_Settings_Page $adminSettingsPage = null)
    {
        $this->restController = $restController ?? new Rest_Controller();
        $this->adminSettingsPage = $adminSettingsPage ?? new Admin_Settings_Page();
    }

    public function init(): void
    {
        if (function_exists('add_action')) {
            add_action('rest_api_init', array($this->restController, 'register_routes'));
            add_action('admin_menu', array($this->adminSettingsPage, 'register_menu'));
            add_action('admin_init', array($this->adminSettingsPage, 'register_settings'));
        }

        if (function_exists('add_shortcode')) {
            add_shortcode(self::SHORTCODE_TAG, array($this, 'render_shortcode'));
        }
    }

    public function render_shortcode(array $atts = array()): string
    {
        $enabled = true;

        if (function_exists('get_option')) {
            $enabled = (bool) get_option(Admin_Settings_Page::OPTION_KEY, true);
        }

        if (! $enabled) {
            $message = '<p>' . $this->escapeHtml($this->translate('Notifications are currently disabled by an administrator.')) . '</p>';

            return $this->wrapMarkup($message, true);
        }

        if (function_exists('wp_enqueue_style')) {
            $assetUrl = $this->getAssetUrl('assets/notifications/shortcode.css');
            if ('' !== $assetUrl) {
                $version = defined('SBDP_VER') ? (string) SBDP_VER : '1.0.0';
                wp_enqueue_style('bsp-notifications-shortcode', $assetUrl, array(), $version);
            }
        }

        $items = array();
        foreach ($this->getSampleNotifications() as $notification) {
            $items[] = sprintf(
                '<li class="bsp-notifications__item"><strong class="bsp-notifications__title">%s</strong><span class="bsp-notifications__description">%s</span></li>',
                $this->escapeHtml($notification['title']),
                $this->escapeHtml($notification['description'])
            );
        }

        $list = sprintf('<ul class="bsp-notifications__list">%s</ul>', implode('', $items));

        $markup = $this->wrapMarkup($list, false);

        if (function_exists('wp_kses_post')) {
            return wp_kses_post($markup);
        }

        return $markup;
    }

    private function wrapMarkup(string $html, bool $disabled): string
    {
        $classes = array('bsp-notifications');
        if ($disabled) {
            $classes[] = 'bsp-notifications--disabled';
        }

        $classAttribute = implode(' ', $classes);
        if (function_exists('esc_attr')) {
            $classAttribute = esc_attr($classAttribute);
        } else {
            $classAttribute = htmlspecialchars($classAttribute, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
        }

        return sprintf('<div class="%s" aria-live="polite">%s</div>', $classAttribute, $html);
    }

    private function getAssetUrl(string $relative): string
    {
        $relativePath = ltrim($relative, '/\\');

        if (defined('SBDP_URL')) {
            return rtrim((string) SBDP_URL, '/\\') . '/' . $relativePath;
        }

        if (function_exists('plugins_url') && defined('SBDP_FILE')) {
            return plugins_url($relativePath, SBDP_FILE);
        }

        return '';
    }

    /**
     * @return array<int, array{title: string, description: string}>
     */
    private function getSampleNotifications(): array
    {
        $time = function_exists('wp_date') ? wp_date('H:i') : date('H:i');

        return array(
            array(
                'title' => sprintf(
                    /* translators: %s: current time string. */
                    $this->translate('Planner sync completed at %s'),
                    $time
                ),
                'description' => $this->translate('All upcoming bookings were synchronised with connected channels.'),
            ),
            array(
                'title' => $this->translate('New booking request submitted'),
                'description' => $this->translate('A guest asked for availability on the "Sunset Canal Tour".'),
            ),
            array(
                'title' => $this->translate('Finance payout pending approval'),
                'description' => $this->translate('Review the payout batch to release partner payments.'),
            ),
        );
    }

    private function translate(string $text): string
    {
        if (function_exists('__')) {
            return __($text, 'sbdp');
        }

        return $text;
    }

    private function escapeHtml(string $value): string
    {
        if (function_exists('esc_html')) {
            return esc_html($value);
        }

        return htmlspecialchars($value, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
    }
}
