```php
declare(strict_types=1);

namespace BSP\Core;

final class CoreServiceProvider
{
    private Modules $modules;

    public function __construct(?Modules $modules = null)
    {
        $this->modules = $modules ?? new Modules();
    }

    public function modules(): Modules
    {
        return $this->modules;
    }

    public function boot(): void
    {
        $modules = $this->modules;

        do_action('bsp/core/modules/register', $modules);

        $modules->boot();

        do_action('bsp/core/modules/booted');
    }
}
```

```php
namespace BSP\Notifications;

use WP_REST_Request;
use WP_REST_Response;
use WP_REST_Server;

final class RestController
{
    public function register(): void
    {
        add_action('rest_api_init', [$this, 'register_routes']);
    }

    public function register_routes(): void
    {
        register_rest_route(
            'bsp/v1',
            '/notifications',
            [
                'methods'             => WP_REST_Server::READABLE,
                'callback'            => [$this, 'get_notifications'],
                'permission_callback' => [$this, 'permissions_check'],
            ]
        );
    }

    public function permissions_check($request): bool
    {
        return current_user_can('read');
    }

    public function get_notifications(WP_REST_Request $request): WP_REST_Response
    {
        return rest_ensure_response(['ok' => true]);
    }
}
```

```php
<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

final class Module implements ModuleInterface
{
    private RestController $rest_controller;

    public function __construct(?RestController $rest_controller = null)
    {
        $this->rest_controller = $rest_controller ?? new RestController();
    }

    public function init(): void
    {
        $this->rest_controller->register();

        add_shortcode('booking_notifications', [$this, 'render_notifications_shortcode']);
        add_action('admin_menu', [$this, 'register_admin_page']);
        add_action('admin_init', [$this, 'register_settings']);
    }

    public function render_notifications_shortcode(array $atts = []): string
    {
        $atts = shortcode_atts(
            [
                'title' => __('Notifications', 'sbdp'),
            ],
            $atts,
            'booking_notifications'
        );

        $title = isset($atts['title']) ? sanitize_text_field($atts['title']) : '';

        $html  = '<div class="bsp-notifications" data-bsp-component="notifications">';
        if ('' !== $title) {
            $html .= '<h3 class="bsp-notifications__title">' . esc_html($title) . '</h3>';
        }
        $html .= '<ul class="bsp-notifications__list">';
        $html .= '<li class="bsp-notifications__item bsp-notifications__item--placeholder">'
            . esc_html__('Loading notifications…', 'sbdp')
            . '</li>';
        $html .= '</ul>';
        $html .= '</div>';

        return (string) wp_kses_post($html);
    }

    public function register_admin_page(): void
    {
        add_options_page(
            __('Booking Pro Notifications', 'sbdp'),
            __('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-notifications',
            [$this, 'render_admin_page']
        );
    }

    public function register_settings(): void
    {
        register_setting(
            'bsp_notifications',
            'bsp_notifications_enabled',
            [
                'type'              => 'boolean',
                'sanitize_callback' => [$this, 'sanitize_enabled_option'],
                'default'           => true,
            ]
        );

        add_settings_section(
            'bsp_notifications_main',
            __('Notification Settings', 'sbdp'),
            '__return_false',
            'bsp_notifications'
        );

        add_settings_field(
            'bsp_notifications_enabled',
            __('Enable notifications', 'sbdp'),
            [$this, 'render_enabled_field'],
            'bsp_notifications',
            'bsp_notifications_main'
        );
    }

    public function render_admin_page(): void
    {
        if (! current_user_can('manage_options')) {
            return;
        }

        ?>
        <div class="wrap">
            <h1><?php echo esc_html__('Booking Pro Notifications', 'sbdp'); ?></h1>
            <form action="options.php" method="post">
                <?php
                settings_fields('bsp_notifications');
                do_settings_sections('bsp_notifications');
                submit_button(__('Save Changes', 'sbdp'));
                ?>
            </form>
        </div>
        <?php
    }

    public function render_enabled_field(): void
    {
        $value = (bool) get_option('bsp_notifications_enabled', true);
        ?>
        <label for="bsp_notifications_enabled">
            <input
                type="checkbox"
                id="bsp_notifications_enabled"
                name="bsp_notifications_enabled"
                value="1"
                <?php checked(true, $value); ?>
            />
            <?php echo esc_html__('Send booking notifications to subscribed contacts.', 'sbdp'); ?>
        </label>
        <?php
    }

    public function sanitize_enabled_option($value): bool
    {
        return (bool) filter_var($value, FILTER_VALIDATE_BOOLEAN);
    }
}
```

```php
namespace BSP\Core\Tests\Notifications;

use BSP\Notifications\Module;
use BSP\Notifications\RestController;
use Brain\Monkey;
use Brain\Monkey\Functions;
use PHPUnit\Framework\TestCase;

final class ModuleTest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();
        Monkey\setUp();
    }

    protected function tearDown(): void
    {
        Monkey\tearDown();
        parent::tearDown();
    }

    public function testInitRegistersHooks(): void
    {
        $hooks = [];
        Functions\when('add_action')->alias(
            static function ($hook, $callback, $priority = 10, $accepted_args = 1) use (&$hooks) {
                $hooks[] = [$hook, $callback, $priority, $accepted_args];
                return true;
            }
        );

        $shortcodes = [];
        Functions\when('add_shortcode')->alias(
            static function ($tag, $callback) use (&$shortcodes) {
                $shortcodes[] = [$tag, $callback];
                return true;
            }
        );

        $module = new Module(new RestController());
        $module->init();

        $hookNames = array_map(static fn($item) => $item[0], $hooks);
        $this->assertContains('rest_api_init', $hookNames);
        $this->assertContains('admin_menu', $hookNames);
        $this->assertContains('admin_init', $hookNames);

        $shortcodeTags = array_map(static fn($item) => $item[0], $shortcodes);
        $this->assertContains('booking_notifications', $shortcodeTags);
    }
```
