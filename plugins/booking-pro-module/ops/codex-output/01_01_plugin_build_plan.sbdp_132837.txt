class Modules

        $this->modules[] = $module;
        $module->init();

    public function boot(): void

}

<?php

declare(strict_types=1);

namespace BSP\Core;

class CoreServiceProvider
{
    private Modules $modules;

    public function __construct(?Modules $modules = null)
    {
        $this->modules = $modules ?? new Modules();
    }

    public function boot(): void
    {
        do_action('bsp/core/modules/register', $this->modules);
        $this->modules->boot();
        do_action('bsp/core/modules/booted', $this->modules);
    }

    public function modules(): Modules
    {
        return $this->modules;
    }
}

<?php

declare(strict_types=1);

namespace BSP\Notifications;

use WP_REST_Controller;
use WP_REST_Request;
use WP_REST_Response;
use WP_REST_Server;

class NotificationsRestController extends WP_REST_Controller
{
    public function __construct()
    {
        $this->namespace = 'bsp/v1';
        $this->rest_base = 'notifications';
    }

    public function register_routes(): void
    {
        register_rest_route(
            $this->namespace,
            '/' . $this->rest_base,
            [
                [
                    'methods' => WP_REST_Server::READABLE,
                    'callback' => [$this, 'get_items'],
                    'permission_callback' => [$this, 'permissions_check'],
                ],
            ]
        );
    }

    public function permissions_check(WP_REST_Request $request)
    {
        return current_user_can('read');
    }

    public function get_items(WP_REST_Request $request): WP_REST_Response
    {
        return new WP_REST_Response(['ok' => true]);
    }
}

<?php

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

class Module implements ModuleInterface
{
    public function init(): void
    {
        add_action('init', [$this, 'register_shortcodes']);
        add_action('rest_api_init', [$this, 'register_rest_routes']);
        add_action('admin_menu', [$this, 'register_admin_page']);
        add_action('admin_init', [$this, 'register_settings']);
    }

    public function boot(): void
    {
        // Module boot logic will be added as features evolve.
    }

    public function register_rest_routes(): void
    {
        $controller = new NotificationsRestController();
        $controller->register_routes();
    }

    public function register_shortcodes(): void
    {
        add_shortcode('booking_notifications', [$this, 'render_notifications_shortcode']);
    }

    public function render_notifications_shortcode(): string
    {
        $title = esc_html__('Booking Notifications', 'sbdp');
        $placeholder = esc_html__('Notifications will appear here.', 'sbdp');

        $html = sprintf(
            '<div class="bsp-notifications"><h2 class="bsp-notifications__title">%1$s</h2><ul class="bsp-notifications__list" data-notifications><li class="bsp-notifications__placeholder">%2$s</li></ul></div>',
            $title,
            $placeholder
        );

        return wp_kses_post($html);
    }

    public function register_admin_page(): void
    {
        add_options_page(
            esc_html__('Booking Pro Suite', 'sbdp'),
            esc_html__('Booking Pro', 'sbdp'),
            'manage_options',
            'bsp-notifications',
            [$this, 'render_settings_page']
        );
    }

    public function render_settings_page(): void
    {
        if (! current_user_can('manage_options')) {
            return;
        }
        ?>
        <div class="wrap">
            <h1><?php echo esc_html__('Booking Pro Suite', 'sbdp'); ?></h1>
            <form method="post" action="options.php">
                <?php
                settings_fields('bsp_notifications');
                do_settings_sections('bsp-notifications');
                submit_button();
                ?>
            </form>
        </div>
        <?php
    }

    public function register_settings(): void
    {
        register_setting(
            'bsp_notifications',
            'bsp_notifications_enabled',
            [
                'type' => 'boolean',
                'sanitize_callback' => [$this, 'sanitize_notifications_toggle'],
                'default' => false,
            ]
        );

        add_settings_section(
            'bsp_notifications_main',
            esc_html__('Notifications Settings', 'sbdp'),
            [$this, 'render_notifications_section'],
            'bsp-notifications'
        );

        add_settings_field(
            'bsp_notifications_enabled',
            esc_html__('Enable notifications', 'sbdp'),
            [$this, 'render_notifications_field'],
            'bsp-notifications',
            'bsp_notifications_main'
        );
    }

    public function render_notifications_section(): void
    {
        echo '<p>' . esc_html__('Control basic notification behavior for Booking Pro Suite.', 'sbdp') . '</p>';
    }

    public function render_notifications_field(): void
    {
        $value = (bool) get_option('bsp_notifications_enabled', false);
        ?>
        <label for="bsp_notifications_enabled">
            <input
                type="checkbox"
                name="bsp_notifications_enabled"
                id="bsp_notifications_enabled"
                value="1"
                <?php checked($value); ?>
            />
            <?php echo esc_html__('Enable planner notifications', 'sbdp'); ?>
        </label>
        <?php
    }

    /**
     * @param mixed $value
     */
    public function sanitize_notifications_toggle($value): bool
    {
        return (bool) $value;
    }
}

mkdir -p dist
rm -f dist/booking-pro-suite.zip

zip -r dist/booking-pro-suite.zip plugin.php core modules assets languages tests vendor

popd >/dev/null
