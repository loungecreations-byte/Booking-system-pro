`core/src/Modules.php`
```php
<?php
declare(strict_types=1);

namespace BSP\Core;

class Modules
{
    /**
     * @var array<string, ModuleInterface>
     */
    private array $modules = [];

    public function register(ModuleInterface $module): void
    {
        $slug = $module->getSlug();
        $this->modules[$slug] = $module;

        $module->init();
    }

    public function boot(): void
    {
        foreach ($this->modules as $module) {
            $module->boot();
        }
    }

    public function has(string $slug): bool
    {
        return isset($this->modules[$slug]);
    }

    /**
     * @return array<string, ModuleInterface>
     */
    public function all(): array
    {
        return $this->modules;
    }
}
```

class CoreServiceProvider

<?php
declare(strict_types=1);

namespace BSP\Notifications;

use WP_REST_Controller;
use WP_REST_Request;
use WP_REST_Response;
use WP_REST_Server;

class NotificationsRestController extends WP_REST_Controller
{
    public function __construct()
    {
        $this->namespace = 'bsp/v1';
        $this->rest_base = 'notifications';
    }

    public function register_routes(): void
    {
        register_rest_route(
            $this->namespace,
            '/' . $this->rest_base,
            [
                [
                    'methods'             => WP_REST_Server::READABLE,
                    'callback'            => [$this, 'get_items'],
                    'permission_callback' => [$this, 'permissions_check'],
                ],
            ]
        );
    }

    public function get_items(WP_REST_Request $request): WP_REST_Response
    {
        return new WP_REST_Response(['ok' => true], 200);
    }

    public function permissions_check(WP_REST_Request $request): bool
    {
        return current_user_can('read');
    }
}

declare(strict_types=1);

namespace BSP\Notifications;

use BSP\Core\ModuleInterface;

class Module implements ModuleInterface
{
    public const SLUG = 'notifications';

    public function getSlug(): string
    {
        return self::SLUG;
    }

    public function init(): void
    {
        add_action('rest_api_init', [$this, 'registerRestRoutes']);
        add_shortcode('booking_notifications', [$this, 'renderNotificationsShortcode']);
    }

    public function boot(): void
    {
        add_action('admin_menu', [$this, 'registerAdminPage']);
        add_action('admin_init', [$this, 'registerSettings']);
    }

    public function registerRestRoutes(): void
    {
        $controller = new NotificationsRestController();
        $controller->register_routes();
    }

    public function renderNotificationsShortcode($atts = [], $content = '', $shortcode = ''): string
    {
        $atts = shortcode_atts(
            [
                'container_id' => 'sbdp-notifications',
                'heading' => __('Booking Notifications', 'sbdp'),
            ],
            $atts,
            $shortcode
        );

        $containerId = $this->sanitizeContainerId((string) $atts['container_id']);
        $heading = esc_html((string) $atts['heading']);

        $html = sprintf(
            '<div class="sbdp-notifications" id="%s"><h3 class="sbdp-notifications__heading">%s</h3><ul class="sbdp-notifications__list" aria-live="polite" aria-busy="false"></ul></div>',
            esc_attr($containerId),
            $heading
        );

        $html = apply_filters('bsp/notifications/shortcode/html', $html, $atts, $shortcode);

        return wp_kses_post($html);
    }

    public function registerAdminPage(): void
    {
        // Admin-pagina volgt in stap 7.
    }

    public function registerSettings(): void
    {
        // Settings-registratie volgt in stap 7.
    }

    private function sanitizeContainerId(string $value): string
    {
        $sanitized = preg_replace('/[^a-zA-Z0-9_-]/', '', $value);

        if ($sanitized === null || $sanitized === '') {
            return 'sbdp-notifications';
        }

        return strtolower($sanitized);
    }
}

declare(strict_types=1);

namespace BSP\Core;

class Modules
{
    /**
     * @var array<string, ModuleInterface>
     */
    private array $modules = [];

    public function register(ModuleInterface $module): void
    {
        $slug = $module->getSlug();
        $this->modules[$slug] = $module;

        $module->init();
    }

    public function boot(): void
    {
        foreach ($this->modules as $module) {
            $module->boot();
        }
    }

    public function has(string $slug): bool
    {
        return isset($this->modules[$slug]);
    }

    /**
     * @return array<string, ModuleInterface>
     */
    public function all(): array
    {
        return $this->modules;
    }
}
